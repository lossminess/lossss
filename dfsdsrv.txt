local function createWaypointVisual(position, index, waitTime, isManaged)
    local part = Instance.new("Part")
    part.Name = "Waypoint_" .. index
    part.Shape = Enum.PartType.Ball
    part.Size = Vector3.new(1.2, 1.2, 1.2)
    part.Position = position
    part.Anchored = true
    part.CanCollide = false
    part.Material = Enum.Material.ForceField
    part.Transparency = 0.2
    if isManaged then
        part.Color = managedWaypointColor
        local billboard = Instance.new("BillboardGui")
        billboard.Name = "WaypointInfo"
        billboard.Size = UDim2.new(0, 80, 0, 30)
        billboard.AlwaysOnTop = true
        billboard.MaxDistance = 150
        billboard.StudsOffset = Vector3.new(0, 2.5, 0)
        billboard.Parent = part
        local textLabel = Instance.new("TextLabel")
        textLabel.Name = "Text"
        textLabel.Size = UDim2.new(1, 0, 1, 0)
        textLabel.BackgroundTransparency = 1
        textLabel.TextColor3 = Color3.new(1, 1, 1)
        textLabel.TextStrokeTransparency = 0.2
        textLabel.TextStrokeColor3 = Color3.new(0, 0, 0)
        textLabel.Font = Enum.Font.SourceSansBold
        textLabel.TextSize = 14
        textLabel.Text = index .. (waitTime and waitTime > 0 and "\n⏱" .. waitTime .. "s" or "")
        textLabel.Parent = billboard
    else
        part.Color = normalWaypointColor
    end
    part.Parent = workspace
    return part
end

local function updateWaypointVisuals()
    for _, part in pairs(waypointParts) do
        if part and part.Parent then
            part:Destroy()
        end
    end
    waypointParts = {}
    local total = #waypoints
    local displayCount = math.min(total, maxDisplayPoints)
    if managedWaypointIndex and managedWaypointIndex > displayCount then
        displayCount = managedWaypointIndex
    end
    for i = 1, math.min(total, displayCount) do
        local waypoint = waypoints[i]
        local isManaged = (i == managedWaypointIndex)
        local part = createWaypointVisual(waypoint.position, i, waypoint.waitTime, isManaged)
        table.insert(waypointParts, part)
    end
end

local function addWaypoint()
    local position = root.Position
    if managedWaypointIndex then
        waypoints[managedWaypointIndex].position = position
        unmanageWaypoint()
        updateWaypointVisuals()
        notify("Waypoint", "Managed waypoint replaced!")
        return
    else
        table.insert(waypoints, { position = position, waitTime = 0 })
        updateWaypointVisuals()
        notify("Waypoint", "New waypoint added!")
    end
end

local function removeAllWaypoints()
    running = false
    waitingInAir = false
    autoSpawnWaypoints = false
    managedWaypointIndex = nil
    normalWaypointColor = defaultWaypointColor
    if autoWalkThread then
        coroutine.close(autoWalkThread)
        autoWalkThread = nil
    end
    if autoSpawnThread then
        coroutine.close(autoSpawnThread)
        autoSpawnThread = nil
    end
    if currentTween then
        currentTween:Cancel()
        currentTween = nil
    end
    waypoints = {}
    updateWaypointVisuals()
    currentWaypointIndex = 1
    if humanoid then
        humanoid:ChangeState(Enum.HumanoidStateType.Running)
    end
    if autoTweenToggle then autoTweenToggle.Text = "OFF" end
    if autoSpawnToggle then autoSpawnToggle.Text = "OFF" end
    notify("Waypoint", "All waypoints removed")
end

local function removeLastWaypoint()
    if #waypoints > 0 then
        if managedWaypointIndex == #waypoints then
            unmanageWaypoint()
        end
        table.remove(waypoints, #waypoints)
        updateWaypointVisuals()
        notify("Waypoint", "Last waypoint removed")
    end
end

local function generateRandomWaypoints()
    removeAllWaypoints()
    local basePos = root.Position
    for i = 1, 20 do
        local randomOffset = Vector3.new(
            math.random(-50, 50),
            math.random(-10, 20),
            math.random(-50, 50)
        )
        local newPos = Vector3.new(
            basePos.X + randomOffset.X,
            basePos.Y + randomOffset.Y,
            basePos.Z + randomOffset.Z
        )
        table.insert(waypoints, {
            position = newPos,
            waitTime = 0
        })
    end
    updateWaypointVisuals()
    notify("Waypoint", "Generated 20 random waypoints")
end

local function manageWaypoint(index)
    if index < 1 or index > #waypoints then
        notify("Waypoint", "Waypoint #" .. index .. " doesn't exist!")
        return false
    end
    if managedWaypointIndex then
        normalWaypointColor = defaultWaypointColor
        managedWaypointIndex = nil
    end
    managedWaypointIndex = index
    normalWaypointColor = Color3.fromRGB(255, 0, 0)
    updateWaypointVisuals()
    notify("Waypoint", "Managing waypoint #" .. index .. " (green)")
    return true
end

local function unmanageWaypoint()
    if not managedWaypointIndex then
        notify("Waypoint", "No waypoint being managed")
        return false
    end
    normalWaypointColor = defaultWaypointColor
    managedWaypointIndex = nil
    updateWaypointVisuals()
    notify("Waypoint", "Stopped managing waypoint")
    return true
end

local function setWaitTime(index, waitTime)
    if waypoints[index] then
        waypoints[index].waitTime = tonumber(waitTime) or 0
        updateWaypointVisuals()
        notify("Waypoint", "Wait time set for waypoint #" .. index)
    else
        notify("Waypoint", "Waypoint #" .. index .. " not found")
    end
end

local function loadWaypointsFromString(input)
    task.spawn(function()
        local data = nil
        local loaded = false
        local success, jsonData = pcall(function()
            return HttpService:JSONDecode(input)
        end)
        if success and type(jsonData) == "table" and #jsonData > 0 then
            data = jsonData
            loaded = true
        end
        if not loaded then
            local temp = {}
            local pattern = "Vector3%.new%s*%(%s*([%-%d%.]+)%s*,%s*([%-%d%.]+)%s*,%s*([%-%d%.]+)%s*%)"
            local waitPattern = "waitTime%s*=%s*([%-%d%.]+)"
            local lines = {}
            for line in input:gmatch("[^\n\r]+") do
                table.insert(lines, line)
            end
            for _, line in ipairs(lines) do
                local x, y, z = line:match(pattern)
                if x and y and z then
                    local wait = 0
                    local w = line:match(waitPattern)
                    if w then
                        wait = tonumber(w) or 0
                    else
                        local commentWait = line:match("wait%s*[:=]%s*([%-%d%.]+)")
                        if commentWait then
                            wait = tonumber(commentWait) or 0
                        end
                    end
                    table.insert(temp, {
                        position = Vector3.new(tonumber(x), tonumber(y), tonumber(z)),
                        waitTime = wait
                    })
                end
            end
            if #temp > 0 then
                data = temp
                loaded = true
            end
        end
        if not loaded then
            local nums = {}
            for num in input:gmatch("[%-%d%.]+") do
                table.insert(nums, tonumber(num))
            end
            if #nums >= 3 then
                local temp = {}
                for i = 1, #nums - 2, 3 do
                    local x, y, z = nums[i], nums[i+1], nums[i+2]
                    if x and y and z then
                        table.insert(temp, {
                            position = Vector3.new(x, y, z),
                            waitTime = 0
                        })
                    end
                end
                if #temp > 0 then
                    data = temp
                    loaded = true
                end
            end
        end
        if loaded and data and #data > 0 then
            running = false
            waitingInAir = false
            autoSpawnWaypoints = false
            managedWaypointIndex = nil
            normalWaypointColor = defaultWaypointColor
            if autoWalkThread then
                coroutine.close(autoWalkThread)
                autoWalkThread = nil
            end
            if autoSpawnThread then
                coroutine.close(autoSpawnThread)
                autoSpawnThread = nil
            end
            if currentTween then
                currentTween:Cancel()
                currentTween = nil
            end
            waypoints = {}
            for _, wp in ipairs(data) do
                local pos = wp.position or (wp.x and wp.y and wp.z and Vector3.new(wp.x, wp.y, wp.z))
                if pos then
                    table.insert(waypoints, {
                        position = pos,
                        waitTime = wp.wait or wp.waitTime or 0
                    })
                end
            end
            currentWaypointIndex = 1
            if humanoid then
                humanoid:ChangeState(Enum.HumanoidStateType.Running)
            end
            if autoTweenToggle then autoTweenToggle.Text = "OFF" end
            if autoSpawnToggle then autoSpawnToggle.Text = "OFF" end
            updateWaypointVisuals()
            notify("Waypoint", "Loaded " .. #waypoints .. " waypoints")
        else
            notify("Waypoint", "Invalid data format")
        end
    end)
end

local function copyWaypoints()
    if #waypoints == 0 then
        notify("Waypoint", "No waypoints to copy")
        return
    end
    local waypointData = {}
    for i, waypoint in ipairs(waypoints) do
        table.insert(waypointData, {
            x = waypoint.position.X,
            y = waypoint.position.Y,
            z = waypoint.position.Z,
            wait = waypoint.waitTime or 0
        })
    end
    local json = HttpService:JSONEncode(waypointData)
    setclipboard(json)
    notify("Waypoint", "Copied " .. #waypoints .. " waypoints to clipboard")
end

local function goToWaypointTween(index)
    if not running or #waypoints == 0 or not waypoints[index] then return false end
    local waypoint = waypoints[index]
    local targetPos = waypoint.position
    local gyro = root:FindFirstChild("WaypointGyro")
    if not gyro then
        gyro = Instance.new("BodyGyro")
        gyro.Name = "WaypointGyro"
        gyro.Parent = root
    end
    gyro.P = 20000
    gyro.D = 500
    gyro.MaxTorque = Vector3.new(400000, 400000, 400000)
    humanoid.AutoRotate = false
    humanoid:ChangeState(Enum.HumanoidStateType.RunningNoPhysics)
    local reached = false
    while running and not reached do
        local currentPos = root.Position
        local distance = (targetPos - currentPos).Magnitude
        if distance < 0.5 then 
            reached = true
            break
        end
        local deltaTime = task.wait()
        local direction = (targetPos - currentPos).Unit
        local moveStep = direction * (tweenSpeed * deltaTime)
        if distance > moveStep.Magnitude then
            root.CFrame = root.CFrame + moveStep
        else
            root.CFrame = CFrame.new(targetPos)
            reached = true
        end
        local lookTarget = Vector3.new(targetPos.X, root.Position.Y, targetPos.Z)
        if (lookTarget - root.Position).Magnitude > 0.1 then
            gyro.CFrame = CFrame.new(root.Position, lookTarget)
        end
        root.Velocity = Vector3.new(0, 0, 0)
    end
    return reached
end

local function goToWaypoint(index)
    if not running or #waypoints == 0 or not waypoints[index] then return end
    local waypoint = waypoints[index]
    local reached = goToWaypointTween(index)
    if running and reached then
        if waypoint.waitTime and waypoint.waitTime > 0 then
            root.CFrame = CFrame.new(waypoint.position)
            root.Velocity = Vector3.new(0,0,0)
            root.Anchored = true
            local startWait = tick()
            while tick() - startWait < waypoint.waitTime and running do
                task.wait(0.1)
            end
            root.Anchored = false
        end
    end
end

local function startAutoWalk()
    if running or #waypoints == 0 then return end
    running = true
    if autoTweenToggle then autoTweenToggle.Text = "ON" end
    autoWalkThread = coroutine.create(function()
        while running and #waypoints > 0 do
            for i = 1, #waypoints do
                if not running or #waypoints == 0 then break end
                currentWaypointIndex = i
                if not root:FindFirstChild("WaypointGyro") then
                    local g = Instance.new("BodyGyro")
                    g.Name = "WaypointGyro"
                    g.P = 10000
                    g.MaxTorque = Vector3.new(400000, 400000, 400000)
                    g.CFrame = root.CFrame
                    g.Parent = root
                end
                goToWaypoint(i)
            end
            task.wait()
        end
        stopAutoWalk()
    end)
    coroutine.resume(autoWalkThread)
end

local function stopAutoWalk()
    running = false
    waitingInAir = false
    if root then
        root.Anchored = false
    end
    if root:FindFirstChild("WaypointGyro") then
        root.WaypointGyro:Destroy()
    end
    humanoid.AutoRotate = true
    if autoWalkThread then
        coroutine.close(autoWalkThread)
        autoWalkThread = nil
    end
    if currentTween then
        currentTween:Cancel()
        currentTween = nil
    end
    if humanoid then
        humanoid:ChangeState(Enum.HumanoidStateType.Running)
    end
    root.Velocity = Vector3.new(0, 0, 0)
    if autoTweenToggle then autoTweenToggle.Text = "OFF" end
end

local function toggleAutoSpawn(interval)
    if autoSpawnWaypoints and autoSpawnThread then
        coroutine.close(autoSpawnThread)
        autoSpawnThread = nil
    end
    if interval > 0 then
        autoSpawnWaypoints = true
        spawnInterval = interval
        if autoSpawnToggle then autoSpawnToggle.Text = "ON" end
        autoSpawnThread = coroutine.create(function()
            while autoSpawnWaypoints do
                addWaypoint()
                task.wait(spawnInterval)
            end
        end)
        coroutine.resume(autoSpawnThread)
    else
        autoSpawnWaypoints = false
        if autoSpawnToggle then autoSpawnToggle.Text = "OFF" end
    end
end

-- ========== Функции Walkspeed и MountainClimber ==========
local function updateWalkspeed()
    if wsCon then wsCon:Disconnect() wsCon = nil end
    if walkspeedEnabled then
        wsCon = RunService.RenderStepped:Connect(function()
            if humanoid then
                humanoid.WalkSpeed = walkspeedValue
            end
        end)
    end
end

local function updateMountainClimber()
    if msaCon then msaCon:Disconnect() msaCon = nil end
    if mountainClimberEnabled then
        msaCon = RunService.RenderStepped:Connect(function()
            if humanoid then
                humanoid.MaxSlopeAngle = 90
            end
        end)
    else
        if humanoid then
            humanoid.MaxSlopeAngle = 46
        end
    end
end

local function createSliderWithLabel(labelText, parent, yPos, minVal, maxVal, defaultVal, step, callback)
    local lbl = createLabel(labelText, parent, yPos)
    local valueDisplay = Instance.new("TextLabel")
    valueDisplay.Size = UDim2.new(0.2, 0, 0, 7)   -- 20 -> 7
    valueDisplay.Position = UDim2.new(0.75, 0, 0, yPos)
    valueDisplay.BackgroundTransparency = 1
    valueDisplay.Text = tostring(defaultVal)
    valueDisplay.TextColor3 = Color3.fromRGB(255, 255, 255)
    valueDisplay.TextSize = 8                   -- 14 -> 8
    valueDisplay.Font = Enum.Font.GothamBold
    valueDisplay.TextXAlignment = Enum.TextXAlignment.Right
    valueDisplay.Parent = parent

    local sliderFrame = Instance.new("Frame")
    sliderFrame.Size = UDim2.new(0.7, 0, 0, 3)   -- 8 -> 3
    sliderFrame.Position = UDim2.new(0.05, 0, 0, yPos + 8) -- 24 -> 8
    sliderFrame.BackgroundColor3 = Color3.fromRGB(80, 80, 90)
    sliderFrame.BorderSizePixel = 0
    local sliderCorner = Instance.new("UICorner")
    sliderCorner.CornerRadius = UDim.new(0, 2)    -- 4 -> 2
    sliderCorner.Parent = sliderFrame
    sliderFrame.Parent = parent

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((defaultVal - minVal) / (maxVal - minVal), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(100, 200, 255)
    fill.BorderSizePixel = 0
    local fillCorner = Instance.new("UICorner")
    fillCorner.CornerRadius = UDim.new(0, 2)      -- 4 -> 2
    fillCorner.Parent = fill
    fill.Parent = sliderFrame

    local sliderBtn = Instance.new("TextButton")
    sliderBtn.Size = UDim2.new(0, 5, 0, 5)        -- 16 -> 5
    sliderBtn.Position = UDim2.new((defaultVal - minVal) / (maxVal - minVal), -3, 0.5, -3) -- -8 -> -3
    sliderBtn.BackgroundColor3 = Color3.fromRGB(200, 200, 210)
    sliderBtn.BorderSizePixel = 0
    sliderBtn.Text = ""
    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 3)       -- 8 -> 3
    btnCorner.Parent = sliderBtn
    sliderBtn.Parent = sliderFrame

    local draggingSlider = false
    sliderBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            draggingSlider = true
        end
    end)
    sliderBtn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            draggingSlider = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if draggingSlider and input.UserInputType == Enum.UserInputType.MouseMovement then
            local absX, absY = sliderFrame.AbsolutePosition.X, sliderFrame.AbsolutePosition.Y
            local mouseX = input.Position.X
            local relX = math.clamp((mouseX - absX) / sliderFrame.AbsoluteSize.X, 0, 1)
            local val = minVal + relX * (maxVal - minVal)
            val = math.round(val / step) * step
            val = math.clamp(val, minVal, maxVal)
            fill.Size = UDim2.new(relX, 0, 1, 0)
            sliderBtn.Position = UDim2.new(relX, -3, 0.5, -3)
            valueDisplay.Text = tostring(val)
            callback(val)
        end
    end)

    return { setValue = function(newVal)
        newVal = math.clamp(newVal, minVal, maxVal)
        local relX = (newVal - minVal) / (maxVal - minVal)
        fill.Size = UDim2.new(relX, 0, 1, 0)
        sliderBtn.Position = UDim2.new(relX, -3, 0.5, -3)
        valueDisplay.Text = tostring(newVal)
    end}
end

    box.Position = UDim2.new(0.05, 0, 0, yPos)
